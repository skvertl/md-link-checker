#### Шаг 2: Реализуйте API вызовы

**Файл (пример, создаётся на этом шаге):** `src/features/new-feature/api/doSomething.ts`

```typescript
import { fetchWithToken } from '@/shared/lib/fetch-mutator';

interface DoSomethingResponse {
  data: unknown;
  status: number;
  headers: Headers;
}

export const doSomething = async (param: string) => {
  const { data } = await fetchWithToken<DoSomethingResponse>(
    `/api/v1/something`,
    {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({ param }),
    }
  );

  return data;
};
```

#### Шаг 3: Реализуйте UI компонент

**Файл (пример, создаётся на этом шаге):** `src/features/new-feature/ui/NewFeatureModal.tsx`

```typescript
import { useState } from 'react';
import { useMutation, useQueryClient } from '@tanstack/react-query';
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
  Input,
  Button,
} from '@/shared/ui';
import { doSomething } from '../api/doSomething';

interface NewFeatureModalProps {
  isOpen: boolean;
  onOpenChange: (open: boolean) => void;
}

export function NewFeatureModal({
  isOpen,
  onOpenChange,
}: NewFeatureModalProps) {
  const [param, setParam] = useState('');
  const queryClient = useQueryClient();

  const doSomethingMutation = useMutation({
    mutationFn: (paramValue: string) => doSomething(paramValue),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['something'] });
      onOpenChange(false);
      setParam('');
    },
    onError: (error) => {
      console.error('Failed to do something', error); // eslint-disable-line no-console
    },
  });

  const handleSave = () => {
    if (!param) return;
    doSomethingMutation.mutate(param);
  };

  return (
    <Dialog open={isOpen} onOpenChange={onOpenChange}>
      <DialogContent className="max-w-xl p-6">
        <DialogHeader>
          <DialogTitle className="text-xl font-semibold">
            Новая фича
          </DialogTitle>
        </DialogHeader>
        <div className="flex items-center gap-4 w-full">
          <Input
            value={param}
            onChange={(e) => setParam(e.target.value)}
            placeholder="Параметр"
            className={`flex-1`}
          />
          <Button
            onClick={handleSave}
            disabled={doSomethingMutation.isPending || !param}
            size="form"
          >
            Сохранить
          </Button>
        </div>
      </DialogContent>
    </Dialog>
  );
}
```

```typescript
import * as React from 'react';
import { cva, type VariantProps } from 'class-variance-authority';
import { cn } from '@/shared/lib';

const newComponentVariants = cva(
  'base-classes',
  {
    variants: {
      variant: {
        default: 'bg-primary text-primary-foreground hover:bg-primary/90',
        secondary: 'bg-secondary text-secondary-foreground hover:bg-secondary/80',
      },
      size: {
        default: 'h-9 px-4 py-2',
        sm: 'h-8 px-3',
        lg: 'h-10 px-6',
      },
    },
    defaultVariants: {
      variant: 'default',
      size: 'default',
    },
  }
);

function NewComponent({
  className,
  variant = 'default',
  size = 'default',
  ...props
}: React.ComponentProps<'button'> &
  VariantProps<typeof newComponentVariants>) {
  return (
    <button
      data-slot="new-component"
      data-variant={variant}
      data-size={size}
      className={cn(newComponentVariants({ variant, size, className }))}
      {...props}
    >
      {props.children}
    </button>
  );
}

// eslint-disable-next-line react-refresh/only-export-components
export { NewComponent, newComponentVariants };
```

```typescript
import { useQuery } from '@tanstack/react-query';
import { fetchWithToken } from '@/shared/lib/fetch-mutator';
import type { NewEntity } from '../model/types';
import { isLocalRun } from '@/shared/lib/environment';
import { getMockNewEntities } from '@/shared/mock/newEntityMocks';

interface NewEntitiesResponse {
  items: NewEntity[];
}

interface NewEntitiesResponseSuccess {
  data: NewEntitiesResponse;
  status: number;
  headers: Headers;
}

export function useNewEntities() {
  return useQuery<NewEntity[]>({
    queryKey: ['new-entities'],
    queryFn: async () => {
      if (isLocalRun()) {
        return getMockNewEntities();
      }

      const { data } = await fetchWithToken<NewEntitiesResponseSuccess>('/new-entities');
      return data.items;
    },
  });
}
```

## Руководство по code review

### Чеклист для Pull Request

**1. Структура и организация кода**
- ✅ Код организован согласно FSD архитектуре
- ✅ Папки и файлы названы корректно (camel-case)
- ✅ Экспорты в `index.ts` файлов
- ✅ Отсутствуют неиспользуемые импорты

**2. Типы и типизация**
- ✅ Все переменные и функции типизированы
- ✅ Используются `interface` или `type` для объектов
- ✅ Отсутствуют `any` типы
- ✅ Используются Zod схемы для валидации данных от API

**3. Тестирование**
- ✅ Новые фичи покрыты тестами
- ✅ Тесты пишутся с Vitest
- ✅ Проверяются позитивные и негативные сценарии
- ✅ Тесты описывают бизнес-логику, а не детали реализации

**4. Производительность**
- ✅ Где уместно, используются оптимистичные обновления UI для мутаций
- ✅ Кэширование в TanStack Query
- ✅ Отсутствуют лишние ре-рендеры (используйте `useMemo`, `useCallback`)

**5. Доступность**
- ✅ Используются ARIA атрибуты
- ✅ Кнопки имеют фокус-состояния
- ✅ Формы имеют метки (`label`, `placeholder`)

**6. Безопасность**
- ✅ Отсутствуют утечки памяти
- ✅ Отсутствуют XSS уязвимости (используйте `dangerouslySetInnerHTML` осторожно)
- ✅ Отсутствуют небезопасные операции `eval`, `innerHTML`

## Распространённые ошибки

### 1. Неправильное использование FSD слоёв

**Ошибка:** Помещение бизнес-логики в слой `shared`

**Решение:** Перенесите бизнес-логику в слой `features` или `entities`

```typescript
// ❌ Неправильно (shared/ui/NewFeatureModal.tsx)
import { useMutation } from '@tanstack/react-query';
// Бизнес-логика в shared слое

// ✅ Правильно (features/new-feature/ui/NewFeatureModal.tsx)
import { useMutation } from '@tanstack/react-query';
// Бизнес-логика в features слое
```

### 2. Пропуск типов TypeScript

**Ошибка:** Использование `any` типов

**Решение:** Определите конкретные типы

```typescript
// ❌ Неправильно
const response = await fetchWithToken('/spaces');
const data: any = response.data;

// ✅ Правильно
import type { Space } from '@/entities/space/model/types';

interface SpacesResponse {
  items: Space[];
}

interface SpacesResponseSuccess {
  data: SpacesResponse;
  status: number;
  headers: Headers;
}

const { data } = await fetchWithToken<SpacesResponseSuccess>('/spaces');
const spaces = data.items;
```

### 3. Отсутствие валидации данных от API

**Ошибка:** Отсутствие Zod валидации данных от API

**Решение:** Используйте Zod схемы для валидации

```typescript
// ❌ Неправильно (без валидации)
const userData = apiResponse.data;

// ✅ Правильно (с валидацией)
import { userDataSchema } from '@/entities/user/model/schema';

const result = userDataSchema.safeParse(apiResponse.data);
if (!result.success) {
  console.error('Validation failed:', result.error);
  throw new Error('Invalid user data');
}

const userData = result.data;
```

### 4. Неправильное использование hooks React

**Ошибка:** Вызов хуков внутри условий или циклов

**Решение:** Хуки должны вызываться только на верхнем уровне компонента

```typescript
// ❌ Неправильно
function Component({ condition }: { condition: boolean }) {
  if (condition) {
    const [count, setCount] = useState(0); // Хук внутри условия
  }
}

// ✅ Правильно
import { useEffect, useState } from 'react';

function Component({ condition }: { condition: boolean }) {
  const [count, setCount] = useState(0); // Хук на верхнем уровне

  useEffect(() => {
    if (condition) {
      setCount(1);
    }
  }, [condition]);
}
```

### 5. Утечки памяти в useEffect

**Ошибка:** Отсутствие очистки в `useEffect`

**Решение:** Используйте `cleanup` функцию

```typescript
// ❌ Неправильно
useEffect(() => {
  const timer = setInterval(() => {
    console.log('Tick');
  }, 1000);
  // Нет очистки таймера
}, []);

// ✅ Правильно
useEffect(() => {
  const timer = setInterval(() => {
    console.log('Tick');
  }, 1000);

  return () => {
    clearInterval(timer); // Очистка таймера
  };
}, []);
```

---

### 6. Неправильное использование `useQuery` ключей

**Ошибка:** Динамические query ключи без зависимости

**Решение:** Используйте зависимость в массиве `queryKey`

```typescript
// ❌ Неправильно
const { data } = useQuery({
  queryKey: ['user', userId], // userId может быть неопределён
  queryFn: () => fetchUser(userId),
});

// ✅ Правильно
const { data } = useQuery({
  queryKey: ['user', userId] as const,
  queryFn: () => fetchUser(userId),
  enabled: !!userId, // Запрос выполняется только если userId определён
});
```
